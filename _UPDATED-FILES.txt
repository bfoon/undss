SET UP, IN ORDER
  1. mkdir secrets
  2. cp un_security_system/.env.example un_security_system/.env
  3. In .env, set ESIGN_SIGNATURE_ID_SECRET now, before anyone enrols:
       python -c "import secrets; print(secrets.token_urlsafe(48))"
  4. Put the office seal at secrets/seal.p12 and its passphrase in .env
     as ESIGN_PADES_PKCS12_PASSPHRASE. For a test certificate only:
       docker compose exec web python manage.py esign_pades \
           --make-dev-cert /app/secrets/seal.p12 --dev-cert-passphrase '...'
  5. docker compose build web && docker compose up -d
  6. docker compose exec web python manage.py makemigrations accounts
     docker compose exec web python manage.py migrate
     docker compose exec web python manage.py esign_identity --install-guard
  7. docker compose exec web python manage.py esign_pades --check
     docker compose exec web python manage.py esign_pades --pdfa-check
     docker compose exec web python manage.py esign_pades --tsa-survey
